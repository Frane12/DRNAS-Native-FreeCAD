cmake_minimum_required(VERSION 3.16)
project(DrnasRenderIsolationTests LANGUAGES C CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
set(CMAKE_AUTOMOC ON)
add_executable(document-open document-open.cpp)
target_link_libraries(document-open PRIVATE Qt6::Widgets Qt6::Test)
target_compile_features(document-open PRIVATE cxx_std_17)
add_executable(surface-lifecycle surface-lifecycle.cpp)
target_link_libraries(surface-lifecycle PRIVATE Qt6::Widgets Qt6::Test)
target_compile_features(surface-lifecycle PRIVATE cxx_std_17)
add_executable(gl-state-isolation gl-state-isolation.c)
target_link_libraries(gl-state-isolation PRIVATE EGL GLESv2 dl)
enable_testing()
add_test(NAME document-open COMMAND document-open)
set_tests_properties(document-open PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME surface-lifecycle COMMAND surface-lifecycle)
set_tests_properties(surface-lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME surface-lifecycle-hidpi COMMAND surface-lifecycle)
set_tests_properties(surface-lifecycle-hidpi PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=2")
