cmake_minimum_required(VERSION 3.22)
project(DrnasDirectTouchTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_executable(direct-touch direct-touch.cpp)
target_link_libraries(direct-touch PRIVATE Qt6::Widgets Qt6::Test)
enable_testing()
add_test(NAME direct-touch COMMAND direct-touch)
set_tests_properties(direct-touch PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(portrait-modal portrait-modal.cpp)
set_target_properties(portrait-modal PROPERTIES AUTOMOC ON)
target_link_libraries(portrait-modal PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME portrait-modal COMMAND portrait-modal)
set_tests_properties(portrait-modal PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

add_test(NAME surface-lifecycle COMMAND surface-lifecycle)
